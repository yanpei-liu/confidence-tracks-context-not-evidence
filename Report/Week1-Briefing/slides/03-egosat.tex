\section{EgoSAT: collision and cross-validation}

\begin{frame}[t]{What EgoSAT is, and where it collides}{\EgosatCite}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      {\footnotesize A streaming egocentric benchmark: \EgosatScale, six tasks spanning retrospective / present / prospective, plus an answerability model and a training-free KV selection method.}
      \vskip0.5em
      {\footnotesize\textbf{Collision surface:}}
      \begin{itemize}\footnotesize\setlength\itemsep{0.1em}
        \item The confidence probe (their Eq.~6 $=$ our definition verbatim).
        \item Confidence vs temporal distance (overlapping question, \EgosatLag\ vs 1--60 min).
        \item Answerability-conditioned confidence $\approx$ our evidence contrast.
        \item Online vs offline (our regime pair, but cross-model not within-backbone).
      \end{itemize}
    \end{column}
    \begin{column}{0.43\textwidth}
      \claim{\footnotesize\color{flagred}\textbf{Must fix in framing}\\[0.3em]
      \begin{itemize}\setlength\itemsep{0.15em}
        \item Drop ``first to evaluate confidence in egocentric streaming.''
        \item Attribute the probe to Tsvilodub et al.\ 2024 / EgoSAT Eq.~6.
        \item Retire ``confidently wrong'' as phrasing.
      \end{itemize}}
      \vskip0.4em
      {\scriptsize\color{inkmid}Position instead on \textbf{evidence age} and the \textbf{elapsed-time-vs-context-volume decomposition}.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[t]{Confidence vs temporal distance: their Table 2 and our $\Delta t$ sweep}{Their conf slope is the raw correlation at \EgosatLag; ours decomposes it over 1--60 min}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      {\footnotesize\textbf{Their finding (Table 2, retrospection):} \EgosatSixOfEight\ confidence slopes are flat or positive. Expected: negative. Their conclusion: ``confidence does not reliably track temporal difficulty.''}
      \vskip0.5em
      {\footnotesize\textbf{What the slope measures.} The raw correlation of mean confidence with retrospective distance (\EgosatLag), across all items. For offline models the context window \textbf{grows} with lag (evidence always present, diluting frames accumulate). For streaming models the cache is bounded, but evidence may be partially evicted. Neither is controlled for token count or evidence presence.}
      \vskip0.5em
      {\footnotesize\textbf{Streaming: qualitative agreement.} Their two streaming models give slopes of \EgosatStreamSlopes; our \Etwo\ also shows flat confidence across $\Delta t$. Both are consistent with ``bounded cache removes the temporal signal.''}
    \end{column}
    \begin{column}{0.48\textwidth}
      {\footnotesize\textbf{Offline: apparent disagreement with \Eone.} Their offline slopes are flat; our elapsed-time contrast is \conf{\EoneElapsedConf}. Three reasons this is not a contradiction:}
      \begin{enumerate}\footnotesize\setlength\itemsep{0.15em}
        \item 45 min of elapsed time vs 16 s --- 170$\times$ the horizon.
        \item Their slope is the \textbf{raw} effect: elapsed time and context growth are confounded. Our token-matched evkeep arm \textbf{isolates} elapsed time --- that decomposition is the contribution.
        \item Their point estimates carry no intervals; half the table sits within rounding distance of zero.
      \end{enumerate}
      \vskip0.3em
      \claim{\scriptsize A difference in estimand, not a contradiction. But it is the first thing a reviewer will cite against \Eone, so the paper should name Table 2 and state the differences.}
    \end{column}
  \end{columns}
\end{frame}
